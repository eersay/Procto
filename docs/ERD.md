# Entity Relationship Diagram

Source of truth: [`backend/prisma/schema.prisma`](../backend/prisma/schema.prisma).

```mermaid
erDiagram
    User ||--o{ Course : "teaches (faculty)"
    User ||--o{ Enrollment : "enrolls"
    User ||--o{ ExamSession : takes
    User ||--o{ RefreshToken : owns
    User ||--o{ AuditLog : generates
    User ||--o{ Announcement : authors
    User ||--o{ Notification : receives

    Course ||--o{ Enrollment : has
    Course ||--o{ Exam : contains
    Course ||--o{ Question : "question bank"
    Course ||--o{ Announcement : has

    Exam ||--|| ExamRules : configures
    Exam ||--o{ ExamQuestion : includes
    Exam ||--o{ ExamSession : "attempted via"

    Question ||--o{ ExamQuestion : "used in"
    Question ||--o{ Answer : "answered by"

    ExamSession ||--o{ Answer : contains
    ExamSession ||--o{ SuspiciousEvent : logs
    ExamSession ||--|| Result : produces

    User {
        string id PK
        string email UK
        string passwordHash
        string googleId UK
        enum role "STUDENT | FACULTY | ADMIN"
        string firstName
        string lastName
        boolean isVerified
        boolean isActive
    }

    Course {
        string id PK
        string code UK "join code students enter"
        string name
        string facultyId FK
    }

    Enrollment {
        string id PK
        string courseId FK
        string studentId FK
        datetime droppedAt "null while active"
    }

    Exam {
        string id PK
        string courseId FK
        string title
        int durationMinutes
        datetime startAt
        datetime endAt
        enum status "DRAFT | SCHEDULED | ACTIVE | COMPLETED | ARCHIVED"
        boolean isPublished
    }

    ExamRules {
        string examId FK
        boolean shuffleQuestions
        boolean shuffleChoices
        int maxAttempts
        float negativeMarkingFactor
        float passThreshold
    }

    Question {
        string id PK
        string courseId FK
        enum type "MULTIPLE_CHOICE | MULTIPLE_SELECT | TRUE_FALSE | SHORT_ANSWER | ESSAY | FILL_BLANK | NUMERICAL | CODE"
        json content "question text, options, correctAnswer"
        float points
    }

    ExamQuestion {
        string examId FK
        string questionId FK
        int orderIndex
    }

    ExamSession {
        string id PK
        string examId FK
        string studentId FK
        enum status "PENDING | ACTIVE | SUBMITTED | TERMINATED | INVALIDATED"
        datetime submittedAt
    }

    Answer {
        string sessionId FK
        string questionId FK
        json response
        float autoScore
        float manualScore
    }

    SuspiciousEvent {
        string sessionId FK
        enum type "FACE_NOT_DETECTED | MULTIPLE_FACES | LOOKING_AWAY | TAB_SWITCH | WINDOW_BLUR | WEBCAM_DENIED | NOISE_DETECTED | SCREEN_EXIT | COPY_PASTE | RIGHT_CLICK"
        enum severity "LOW | MEDIUM | HIGH"
        datetime timestamp
    }

    Result {
        string sessionId FK
        float totalScore
        float percentage
        boolean passStatus
        boolean isPublished
    }

    Notification {
        string userId FK
        string type
        string title
        boolean isRead
    }
```

## Notes

- **`Course.code`** doubles as the join code students type to enroll (format
  `xxx-xxxx-xxx`, generated by [`utils/courseCode.ts`](../backend/src/utils/courseCode.ts)) —
  it's not a department code like "CS201".
- **`Question.content`** and **`Answer.response`** are `Json` columns; their
  shape depends on `Question.type` (see [`utils/grading.ts`](../backend/src/utils/grading.ts)
  for exactly what each question type expects).
- Soft deletes (`deletedAt`) are used on `User`, `Course`, `Exam`, and
  `Question` rather than hard deletes, so historical exam sessions and results
  stay intact even if a course or question is later removed.
